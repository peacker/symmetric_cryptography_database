cmake_minimum_required(VERSION 3.16)
project(Iphe_512_reference C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(iphe_algorithm STATIC CryptHash_AlgorithmInstance.c)
target_include_directories(iphe_algorithm PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(iphe_selftest iphe_selftest.c)
target_link_libraries(iphe_selftest PRIVATE iphe_algorithm)
add_executable(KAT_CryptHash KAT_CryptHash.c drng.c)
target_link_libraries(KAT_CryptHash PRIVATE iphe_algorithm)
enable_testing()
add_test(NAME iphe_selftest COMMAND iphe_selftest)
